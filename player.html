const params = new URLSearchParams(window.location.search); const rawUrl = decodeURIComponent(params.get("url") || ""); const title = params.get("title") || "Video Player"; const video = document.getElementById("player"); const titleEl = document.getElementById("title"); const infoEl = document.getElementById("info"); const dlBtn = document.getElementById("dlbtn"); const proxyBtn = document.getElementById("proxybtn"); if (!video || !titleEl || !infoEl || !dlBtn || !proxyBtn) { console.error("Required HTML elements not found."); } else { titleEl.textContent = title; document.title = title; if (!rawUrl) { titleEl.textContent = "No video URL provided!"; infoEl.textContent = "Add ?url=YOUR_VIDEO_LINK"; } else { const proxyUrl = "https://corsproxy.io/?" + encodeURIComponent(rawUrl); dlBtn.href = rawUrl; proxyBtn.href = "https://www.hlsplayer.net/play?url=" + encodeURIComponent(rawUrl); let proxyTried = false; function loadVideo(url) { video.src = url; video.load(); video.play().catch(() => { console.log("Autoplay blocked"); }); } loadVideo(rawUrl); video.onerror = () => { if (!proxyTried) { proxyTried = true; infoEl.textContent = "Direct stream failed. Trying proxy..."; loadVideo(proxyUrl); } else { infoEl.textContent = "Playback failed. Use Download button."; } }; video.onloadeddata = () => { infoEl.textContent = "Video loaded successfully"; }; } }
