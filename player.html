const params = new URLSearchParams(window.location.search); const rawUrl = params.get("url"); const title = params.get("title") || "Video Player"; const video = document.getElementById("player"); const titleEl = document.getElementById("title"); const infoEl = document.getElementById("info"); const dlBtn = document.getElementById("dlbtn"); const proxyBtn = document.getElementById("proxybtn"); titleEl.textContent = title; document.title = title; if (!rawUrl) { titleEl.textContent = "No video URL provided!"; infoEl.textContent = "Add ?url=YOUR_VIDEO_LINK to the URL"; } else { const proxyUrl = "https://corsproxy.io/?" + encodeURIComponent(rawUrl); dlBtn.href = rawUrl; proxyBtn.href = "https://www.hlsplayer.net/play?url=" + encodeURIComponent(rawUrl); function loadVideo(url) { video.src = url; video.load(); const playPromise = video.play(); if (playPromise !== undefined) { playPromise.catch(() => { console.log("Autoplay blocked by browser."); }); } } // Load direct source first loadVideo(rawUrl); // If loading fails, try proxy video.onerror = function () { infoEl.textContent = "Direct source failed. Trying proxy..."; video.onerror = function () { infoEl.textContent = "Playback failed. Use Download or Proxy Player."; }; loadVideo(proxyUrl); }; }
